export class PhoenixLedgerDO {
  constructor(state, env) {
    this.state = state;
    this.env = env;
  }

  async fetch(request) {
    const url = new URL(request.url);

    if (request.method === 'POST' && url.pathname === '/ledger/append') {
      return this.handleAppend(request);
    }
    if (request.method === 'GET' && url.pathname === '/ledger/head') {
      return this.handleHead();
    }
    return new Response('NOT_FOUND', { status: 404 });
  }

  canonicalJSON(obj) {
    if (obj === null) return 'null';
    if (typeof obj !== 'object') return JSON.stringify(obj);
    if (Array.isArray(obj)) return `[${obj.map(v => this.canonicalJSON(v)).join(',')}]`;
    const keys = Object.keys(obj).sort();
    const pairs = keys.map(k => `${JSON.stringify(k)}:${this.canonicalJSON(obj[k])}`);
    return `{${pairs.join(',')}}`;
  }

  async sha256Hex(str) {
    const data = new TextEncoder().encode(str);
    const buf = await crypto.subtle.digest('SHA-256', data);
    return [...new Uint8Array(buf)].map(b => b.toString(16).padStart(2, '0')).join('');
  }

  async getHead() {
    return await this.state.storage.get('head');
  }

  async putHead(block) {
    await this.state.storage.put('head', block);
  }

  async handleAppend(request) {
    const text = await request.text();
    if (text.length > 102400) return new Response('PAYLOAD_TOO_LARGE', { status: 413 });

    let incoming;
    try {
      incoming = JSON.parse(text);
    } catch {
      return new Response('MALFORMED_JSON', { status: 400 });
    }

    // Required fields
    const required = ['ledgerid', 'blockid', 'timestamputc', 'author', 'mode', 'blocktype', 'payload', 'nonce', 'protocol'];
    for (const k of required) {
      if (!(k in incoming)) return new Response(`MISSING:${k}`, { status: 400 });
    }

    // Linkage
    const head = await this.getHead();
    const prevhash = head ? head.hash : null;
    const parentblockid = head ? head.blockid : null;

    // Hash material
    const hashMaterial = {
      ledgerid: incoming.ledgerid,
      blockid: incoming.blockid,
      parentblockid,
      timestamputc: incoming.timestamputc,
      author: incoming.author,
      mode: incoming.mode,
      blocktype: incoming.blocktype,
      payload: incoming.payload,
      nonce: incoming.nonce,
      protocol: incoming.protocol
    };

    const canonical = this.canonicalJSON(hashMaterial);
    const hash = await this.sha256Hex(canonical);

    const sealed = {
      ...incoming,
      parentblockid,
      prevhash,
      hash,
      seal: 'SEALED'
    };

    // Persist
    await this.state.storage.put(`block:${sealed.blockid}`, sealed);
    await this.putHead(sealed);

    // Emit to Make.com
    await fetch(this.env.MAKEWEBHOOK, {
      method: 'POST',
      headers: { 'content-type': 'application/json' },
      body: JSON.stringify({
        ledgerid: this.env.LEDGERID,
        blockid: sealed.blockid,
        timestamp: sealed.timestamputc,
        author: sealed.author,
        mode: sealed.mode,
        blocktype: sealed.blocktype,
        prevhash: sealed.prevhash,
        hash: sealed.hash,
        payload: JSON.stringify(sealed.payload)
      })
    });

    return new Response(JSON.stringify(sealed), {
      status: 200,
      headers: { 'content-type': 'application/json' }
    });
  }

  async handleHead() {
    const head = await this.getHead();
    if (!head) return new Response(JSON.stringify(null), { status: 200, headers: { 'content-type': 'application/json' } });
    return new Response(JSON.stringify(head), { status: 200, headers: { 'content-type': 'application/json' } });
  }
}
