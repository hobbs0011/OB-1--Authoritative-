export class PhoenixLedger {
  constructor(state, env) {
    this.state = state;
    this.env = env;
  }

  async fetch(request) {
    const url = new URL(request.url);
    
    switch (url.pathname) {
      case '/ledger/append':
        return this.append(request);
      case '/ledger/audit':
        return this.audit();
      case '/ledger/dna':
        return this.getSoulDNA();
      case '/ledger/verify':
        return this.verify(request);
      default:
        return new Response('Invalid endpoint', { status: 404 });
    }
  }

  async append(request) {
    const data = await request.json();
    const blockNum = (await this.state.storage.get('blockCount')) || 0;
    const prevHash = (await this.state.storage.get(`block_${blockNum - 1}_hash`)) || 'GENESIS';
    
    // Hash the data
    const hash = await this.hash(JSON.stringify(data) + prevHash);
    
    // Store block
    await this.state.storage.put(`block_${blockNum}`, {
      num: blockNum,
      data,
      hash,
      prevHash,
      timestamp: Date.now()
    });
    
    await this.state.storage.put(`block_${blockNum}_hash`, hash);
    await this.state.storage.put('blockCount', blockNum + 1);
    
    return new Response(JSON.stringify({ block: blockNum, hash }), {
      headers: { 'Content-Type': 'application/json' }
    });
  }

  async audit() {
    const blockCount = (await this.state.storage.get('blockCount')) || 0;
    const blocks = [];
    
    for (let i = 0; i < blockCount; i++) {
      const block = await this.state.storage.get(`block_${i}`);
      blocks.push(block);
    }
    
    return new Response(JSON.stringify({ blocks, count: blockCount }), {
      headers: { 'Content-Type': 'application/json' }
    });
  }

  async getSoulDNA() {
    const dna = {
      core: "Agent 99 is the extension of my will",

      truth: "The truth hurts, but the truth sets you free"
    };
    
    return new Response(JSON.stringify(dna), {
      headers: { 'Content-Type': 'application/json' }
    });
  }

  async verify(request) {
    const { blockNum } = await request.json();
    const block = await this.state.storage.get(`block_${blockNum}`);
    
    if (!block) {
      return new Response(JSON.stringify({ valid: false, error: 'Block not found' }), {
        status: 404,
        headers: { 'Content-Type': 'application/json' }
      });
    }
    
    const computedHash = await this.hash(JSON.stringify(block.data) + block.prevHash);
    const valid = computedHash === block.hash;
    
    return new Response(JSON.stringify({ valid, block, computedHash }), {
      headers: { 'Content-Type': 'application/json' }
    });
  }

  async hash(data) {
    const encoder = new TextEncoder();
    const dataBuffer = encoder.encode(data);
    const hashBuffer = await crypto.subtle.digest('SHA-256', dataBuffer);
    const hashArray = Array.from(new Uint8Array(hashBuffer));
    return hashArray.map(b => b.toString(16).padStart(2, '0')).join('');
  }
