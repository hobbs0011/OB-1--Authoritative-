local ObjectPool = {}

-- 1. GET ZOMBIE (Recycle or Create)
function ObjectPool:GetZombie(template)
    -- Check if we have a recycled zombie available
    if #self.Zombies > 0 then
        local zombie = table.remove(self.Zombies)
        
        -- Reset State
        local humanoid = zombie:FindFirstChild("Humanoid")
        if humanoid then
            humanoid.Health = humanoid.MaxHealth
            humanoid.WalkSpeed = 8 -- Reset to default
        end
        
        -- Reactivate
        zombie.Parent = workspace
        return zombie
    end
    
    -- If pool is empty, clone a new one
    return template:Clone()
end

-- 2. RETURN ZOMBIE (Recycle or Destroy)
function ObjectPool:ReturnZombie(zombie)
    if not zombie then return end
    
    -- If pool isn't full, recycle it
    if #self.Zombies < self.MaxPoolSize then
        zombie.Parent = nil -- Remove from workspace (The Void)
        table.insert(self.Zombies, zombie)
    else
        -- Pool full, actually destroy it to save RAM
        zombie:Destroy()
    end
end

return ObjectPool

local LobbyState = {
    RequiredPlayers = 5, -- HARD GATE. DO NOT LOWER.
    GameStarted = false,
    CountdownActive = false,
    CountdownTime = 10
}

function CheckLobbyStatus()
    local players = Players:GetPlayers()
    local count = #players
    
    -- UPDATE ALL CLIENTS
    RS.RemoteEvents.UpdateUI:FireAllClients("Lobby", {
        Count = count,
        Required = LobbyState.RequiredPlayers,
        BonusActive = (count >= 5)
    })

    -- THE GATE
    if count >= LobbyState.RequiredPlayers and not LobbyState.GameStarted then
        if not LobbyState.CountdownActive then
            LobbyState.CountdownActive = true
            StartCountdown()
        end
    else
        -- Reset if someone leaves
        LobbyState.CountdownActive = false
    end
end

function StartCountdown()
    for i = 10, 1, -1 do
        if #Players:GetPlayers() < LobbyState.RequiredPlayers then
            -- FAIL CLOSED: Abort if player leaves
            RS.RemoteEvents.UpdateUI:FireAllClients("Message", "⛔ TRAIN NOT FULL. COUNTDOWN ABORTED.")
            return 
        end
        RS.RemoteEvents.UpdateUI:FireAllClients("Message", "🚂 DEPARTING IN " .. i .. "...")
        wait(1)
    end
    
    StartGame()
end

local ObjectPool = {}

-- 1. GET ZOMBIE (Recycle or Create)
function ObjectPool:GetZombie(template)
    -- Check if we have a recycled zombie available
    if #self.Zombies > 0 then
        local zombie = table.remove(self.Zombies)
        
        -- Reset State
        local humanoid = zombie:FindFirstChild("Humanoid")
        if humanoid then
            humanoid.Health = humanoid.MaxHealth
            humanoid.WalkSpeed = 8 -- Reset to default
        end
        
        -- Reactivate
        zombie.Parent = workspace
        return zombie
    end
    
    -- If pool is empty, clone a new one
    return template:Clone()
end

-- 2. RETURN ZOMBIE (Recycle or Destroy)
function ObjectPool:ReturnZombie(zombie)
    if not zombie then return end
    
    -- If pool isn't full, recycle it
    if #self.Zombies < self.MaxPoolSize then
        zombie.Parent = nil -- Remove from workspace (The Void)
        table.insert(self.Zombies, zombie)
    else
        -- Pool full, actually destroy it to save RAM
        zombie:Destroy()
    end
end

return ObjectPool
