// =======================================
// STAGE 3 — RUNTIME ENGINE
// Requires Stage 1 + Stage 2 loaded
// =======================================

(function () {
  if (typeof CommandStation === "undefined") {
    throw new Error("CommandStation Stage 1 must be loaded before Stage 3.");
  }

  const cs = CommandStation;

  const nowISO = () => new Date().toISOString();

  function safeLog(...args) {
    if (cs.config && cs.config.allowDebug) {
      try {
        console.log("[CommandStation]", ...args);
      } catch (e) {}
    }
  }

  // Ensure runtime namespace
  cs.runtime = cs.runtime || {};

  // =======================================
  // EVENT BUS
  // =======================================

  const Bus = {
    listeners: {},

    on(event, fn) {
      if (!Bus.listeners[event]) Bus.listeners[event] = [];
      Bus.listeners[event].push(fn);
      return () => Bus.off(event, fn);
    },

    off(event, fn) {
      if (!Bus.listeners[event]) return;
      Bus.listeners[event] = Bus.listeners[event].filter(x => x !== fn);
    },

    emit(event, payload) {
      if (!Bus.listeners[event]) return;
      for (const fn of Bus.listeners[event]) {
        try {
          fn(payload);
        } catch (err) {
          safeLog("Bus listener error:", err);
        }
      }
    }
  };

  // =======================================
  // COMMAND REGISTRY
  // =======================================

  const Commands = {
    registry: {},

    register(name, fn) {
      if (Commands.registry[name]) {
        safeLog(`Command overwritten: ${name}`);
      }
      Commands.registry[name] = fn;
      return fn;
    },

    async run(name, ...args) {
      const fn = Commands.registry[name];
      if (!fn) throw new Error(`Command not found: ${name}`);
      Bus.emit("command:before", { name, args, at: nowISO() });
      const result = await fn(...args);
      Bus.emit("command:after", { name, result, at: nowISO() });
      return result;
    }
  };

  // =======================================
  // TASK SCHEDULER
  // =======================================

  const Tasks = {
    tasks: {},

    schedule(name, intervalMs, fn, opts = {}) {
      if (Tasks.tasks[name]) {
        safeLog(`Task overwritten: ${name}`);
        clearInterval(Tasks.tasks[name].handler);
      }

      const handler = setInterval(async () => {
        try {
          Bus.emit("task:tick", { name, at: nowISO() });
          await fn();
        } catch (err) {
          Bus.emit("task:error", { name, err, at: nowISO() });
        }
      }, intervalMs);

      Tasks.tasks[name] = { name, intervalMs, fn, handler, opts };
      Bus.emit("task:scheduled", { name, intervalMs, at: nowISO() });
      return name;
    },

    cancel(name) {
      if (!Tasks.tasks[name]) return false;
      clearInterval(Tasks.tasks[name].handler);
      delete Tasks.tasks[name];
      Bus.emit("task:cancelled", { name, at: nowISO() });
      return true;
    }
  };

  // =======================================
  // PREBUILT DEFAULT COMMANDS
  // =======================================

  Commands.register("echo", msg => msg);

  Commands.register("time", () => nowISO());

  Commands.register("config:get", key => {
    return key ? cs.config[key] : cs.config;
  });

  Commands.register("config:set", (key, value) => {
    cs.config[key] = value;
    return true;
  });

  Commands.register("env:get", key => {
    return key ? cs.env[key] : cs.env;
  });

  // =======================================
  // EXPOSE RUNTIME
  // =======================================

  cs.runtime.bus = Bus;
  cs.runtime.commands = Commands;
  cs.runtime.tasks = Tasks;

  cs.env.stage3BootedAt = nowISO();
  safeLog("Stage 3 runtime initialized.");
})();
