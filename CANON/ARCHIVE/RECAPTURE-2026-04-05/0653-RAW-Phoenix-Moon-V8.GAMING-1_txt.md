THE PROBLEM: Instance.new("Model") and :Destroy() are expensive CPU operations. On a low-end mobile device (iPhone 8 / Android), spawning and destroying 20 zombies every minute causes "Garbage Collection Spikes" (Lag).

local ObjectPool = {}
ObjectPool.Zombies = {}
ObjectPool.MaxPoolSize = 50 -- Cap memory usage

-- 1. GET ZOMBIE (Recycle or Create)
function ObjectPool:GetZombie(template)
    -- Check if we have a recycled zombie available
    if #self.Zombies > 0 then
        local zombie = table.remove(self.Zombies)
        
        -- Reset State
        local humanoid = zombie:FindFirstChild("Humanoid")
        if humanoid then
            humanoid.Health = humanoid.MaxHealth
            humanoid.WalkSpeed = 8 -- Reset to default
        end
        
        -- Reactivate
        zombie.Parent = workspace
        return zombie
    end
    
    -- If pool is empty, clone a new one
    return template:Clone()
end

-- 2. RETURN ZOMBIE (Recycle or Destroy)
function ObjectPool:ReturnZombie(zombie)
    if not zombie then return end
    
    -- If pool isn't full, recycle it
    if #self.Zombies < self.MaxPoolSize then
        zombie.Parent = nil -- Remove from workspace (The Void)
        table.insert(self.Zombies, zombie)
    else
        -- Pool full, actually destroy it to save RAM
        zombie:Destroy()
    end
end

return ObjectPool
