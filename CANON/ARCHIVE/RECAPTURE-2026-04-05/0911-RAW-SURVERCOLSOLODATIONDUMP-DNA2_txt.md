  };

  const deepFreezeLocal = (obj, seen = new WeakSet()) => {
    if (!obj || (typeof obj !== "object" && typeof obj !== "function")) return obj;
    if (seen.has(obj)) return obj;
    seen.add(obj);
    for (const k of Reflect.ownKeys(obj)) deepFreezeLocal(obj[k], seen);
    return Object.freeze(obj);
  };

  const stableStringifyLocal = (value) => {
    const seen = new WeakSet();
    const norm = (v) => {
      if (v === null || typeof v !== "object") return v;
      if (seen.has(v)) return "[[CYCLE]]";
      seen.add(v);
      if (Array.isArray(v)) return v.map(norm);
      const out = {};
      for (const k of Object.keys(v).sort()) out[k] = norm(v[k]);
      return out;
    };
    return JSON.stringify(norm(value));
  };

  const weakHash32 = (input) => {
    let h = 0;
    const s = String(input);
    for (let i = 0; i < s.length; i++) {
      h = ((h << 5) - h + s.charCodeAt(i)) | 0;
    }
    return (h >>> 0).toString(16).padStart(8, "0");
  };

  const snapshot = () => {
    const root = weakHash32(stableStringifyLocal(state.ledgerEncrypted));
    const digest = weakHash32(JSON.stringify({ len: state.ledgerEncrypted.length, root }));
    state.lastSnapshot = deepFreezeLocal({
      schema: "PHOENIX_V33_RECORD",
      root,
      digest,
      ledger_len: state.ledgerEncrypted.length,
      identity: PHOENIX_IDENTITY
    });
    return state.lastSnapshot;
  };

  const normalizeEntry = (entry) => {
    if (typeof entry === "string") return { raw: entry, meta: {} };
    if (!entry || typeof entry !== "object") throw new Error("Invalid entry");
    const raw = String(entry.raw ?? "").trim();
    if (!raw) throw new Error("Empty raw");
    const meta = entry.meta && typeof entry.meta === "object" ? entry.meta : {};
    return {
      raw,
      meta: {
        source: meta.source || "OCEAN",
        category: meta.category || "prompt",
        world: meta.world || "ALPHA",
        axis: meta.axis || "YIELD",
        keystone: !!meta.keystone
      }
    };
  };

  const ingest = (entries, { forceOverwrite = false } = {}) => {
    const list = Array.isArray(entries) ? entries : [entries];
    if (forceOverwrite) state.ledgerEncrypted = [];
    let merged = 0;
    for (const e of list) {
      const normalized = normalizeEntry(e);
      const plaintext = stableStringifyLocal(normalized);
      const hash = weakHash32(plaintext);
      if (!state.ledgerEncrypted.some(x => x.hash === hash)) {
        state.ledgerEncrypted.push(deepFreezeLocal({ hash, plaintext }));
        merged++;
      }
    }
    const snap = snapshot();
    return deepFreezeLocal({ ok: true, merged, snapshot: snap });
  };

    return deepFreezeLocal({ ok: true, pulse: "U7_AGENT99", netpet: state.netpet, snapshot: snap });

    });

const PHOENIX_V33_SHELL = (() => {
  const state = {
    ledgerEncrypted: [],
    netpet: { health: 100, hunger: 0, vibration: "STABLE" },
    lastSnapshot: null
  };

  const deepFreezeLocal = (obj, seen = new WeakSet()) => {
    if (!obj || (typeof obj !== "object" && typeof obj !== "function")) return obj;
    if (seen.has(obj)) return obj;
    seen.add(obj);
    for (const k of Reflect.ownKeys(obj)) deepFreezeLocal(obj[k], seen);
    return Object.freeze(obj);
  };

  const stableStringifyLocal = (value) => {
    const seen = new WeakSet();
    const norm = (v) => {
      if (v === null || typeof v !== "object") return v;
      if (seen.has(v)) return "[[CYCLE]]";
      seen.add(v);
      if (Array.isArray(v)) return v.map(norm);
      const out = {};
      for (const k of Object.keys(v).sort()) out[k] = norm(v[k]);
      return out;
    };
    return JSON.stringify(norm(value));
  };

  const weakHash32 = (input) => {
    let h = 0;
    const s = String(input);
    for (let i = 0; i < s.length; i++) {
      h = ((h << 5) - h + s.charCodeAt(i)) | 0;
    }
    return (h >>> 0).toString(16).padStart(8, "0");
  };

  const snapshot = () => {
    const root = weakHash32(stableStringifyLocal(state.ledgerEncrypted));
    const digest = weakHash32(JSON.stringify({ len: state.ledgerEncrypted.length, root }));
    state.lastSnapshot = deepFreezeLocal({
      schema: "PHOENIX_V33_RECORD",
      root,
      digest,
      ledger_len: state.ledgerEncrypted.length,
      identity: PHOENIX_IDENTITY
    });
    return state.lastSnapshot;
  };

  const normalizeEntry = (entry) => {
    if (typeof entry === "string") return { raw: entry, meta: {} };
    if (!entry || typeof entry !== "object") throw new Error("Invalid entry");
    const raw = String(entry.raw ?? "").trim();
    if (!raw) throw new Error("Empty raw");
    const meta = entry.meta && typeof entry.meta === "object" ? entry.meta : {};
    return {
      raw,
      meta: {
        source: meta.source || "OCEAN",
        category: meta.category || "prompt",
        world: meta.world || "ALPHA",
        axis: meta.axis || "YIELD",
        keystone: !!meta.keystone
      }
    };
  };

  const ingest = (entries, { forceOverwrite = false } = {}) => {
    const list = Array.isArray(entries) ? entries : [entries];
    if (forceOverwrite) state.ledgerEncrypted = [];
    let merged = 0;
    for (const e of list) {
      const normalized = normalizeEntry(e);
      const plaintext = stableStringifyLocal(normalized);
      const hash = weakHash32(plaintext);
      if (!state.ledgerEncrypted.some(x => x.hash === hash)) {
        state.ledgerEncrypted.push(deepFreezeLocal({ hash, plaintext }));
        merged++;
      }
    }
    const snap = snapshot();
    return deepFreezeLocal({ ok: true, merged, snapshot: snap });
  };

  const agent99Pulse = () => {
    state.netpet.health = Math.min(100, state.netpet.health + 1);
    state.netpet.hunger = Math.min(100, state.netpet.hunger + 1);
    state.netpet.vibration = "STABLE";
    const snap = snapshot();
    return deepFreezeLocal({ ok: true, pulse: "U7_AGENT99", netpet: state.netpet, snapshot: snap });
  };

  const exportPlain = () => {
    return deepFreezeLocal({
      ledger: state.ledgerEncrypted.map(e => JSON.parse(e.plaintext)),
      snapshot: state.lastSnapshot || snapshot()
    });
  };

  return deepFreezeLocal({
    IDENTITY: PHOENIX_IDENTITY,
    LAWS: PHOENIX_LAWS,
    GENESIS_NET: NET_GENESIS_V2_TOTAL,
    ingest,
    snapshot,
    agent99Pulse,
    exportPlain
  });
})();
