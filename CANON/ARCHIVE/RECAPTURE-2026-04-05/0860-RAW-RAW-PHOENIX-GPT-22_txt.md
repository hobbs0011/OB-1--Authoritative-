## **MODULE: receipts.js**

"use strict";

// Receipt after successful EXECUTION
export async function writeReceipt(doState, block, result) {
  const receipt = Object.freeze({
    receipt_id: `R-${block.block_id}`,
    block_id: block.block_id,
    block_hash: block.hash,
    execution_status: result.status, // SUCCESS|FAILURE
    timestamp_utc: new Date().toISOString(),
    result_hash: await hashResult(result),
    side_effects: result.side_effects || []
  });
  
  await doState.storage.put(`receipt:${receipt.receipt_id}`, receipt);
  return receipt;
}

// Check if already executed (idempotency)
export async function checkReceipt(doState, block_id) {
  return await doState.storage.get(`receipt:R-${block_id}`);
}

// Orphan detection
export async function isOrphan(doState, parent_block_id) {
  if (!parent_block_id) return false; // Genesis
  const parent = await doState.storage.get(`block:${parent_block_id}`);
  return !parent;
}

// Quarantine orphan (readable, doesn't advance head)
export async function quarantineOrphan(doState, block) {
  await doState.storage.put(`orphan:${block.block_id}`, {
    ...block,
    quarantined_at: new Date().toISOString(),
    reason: "MISSING_PARENT"
  });
}

async function hashResult(result) {
  const str = JSON.stringify(result);
  const buf = await crypto.subtle.digest("SHA-256", new TextEncoder().encode(str));
  return [...new Uint8Array(buf)].map(b => b.toString(16).padStart(2, "0")).join("");
}

export function stableStringify(v) {
  if (v === null) return "null";
  const t = typeof v;
  if (t === "number" || t === "boolean") return JSON.stringify(v);
  if (t === "string") return JSON.stringify(v);
  if (Array.isArray(v)) return "[" + v.map(stableStringify).join(",") + "]";
  if (t === "object") {
    const keys = Object.keys(v).sort();
    return "{" + keys.map(k => JSON.stringify(k) + ":" + stableStringify(v[k])).join(",") + "}";
  }
  return JSON.stringify(String(v));
}

export function stableStringify(value) {
  if (value === null) return "null";

  const t = typeof value;
  if (t === "number" || t === "boolean" || t === "string") {
    return JSON.stringify(value);
  }

  if (Array.isArray(value)) {
    const inner = value.map(stableStringify).join(",");
    return "[" + inner + "]";

      parts.push(JSON.stringify(k) + ":" + stableStringify(v));

  scope,

// GEM 71: Shallow clone helper (secure)
function shallowCloneWithout(obj, key) {
  if (!obj || typeof obj !== "object") return {};
  const clone = {};
  for (const k of Object.keys(obj)) {
    if (k === key) continue;
    clone[k] = obj[k];
  }
  return clone;
}

// GEM 72: Head proof verifier
export async function verifyHeadProof(proof) {
  if (!proof || proof.type !== "HEAD_PROOF") {
    return { ok: false, reason: "NOT_HEAD_PROOF" };
  }
  
  try {
    requireHex64("head_hash", proof.head_hash);
    requireHex64("merkle_root", proof.merkle_root);
  } catch (e) {
    return { ok: false, reason: e.code || "INVALID_PROOF_FIELDS" };
  }
  
  const body = shallowCloneWithout(proof, "proof_hash");
  const recomputed = await sha256HexCanonical(body);
  
  if (recomputed !== proof.proof_hash) {
    return { ok: false, reason: "PROOF_HASH_MISMATCH" };
  }
  
  return { ok: true };
}

// GEM 80: Quorum validator with deduplication
export function validateQuorumApprovals(approvals, required_count) {
  if (!Array.isArray(approvals)) {
    throw new QuorumError("APPROVALS_MUST_BE_ARRAY", { approvals });
  }
  
  // GEM 81: Approval count validation
  if (approvals.length > PHOENIX.LIMITS.MAX_APPROVALS) {
    throw new QuorumError("TOO_MANY_APPROVALS", { count: approvals.length });
  }
  
  // GEM 82: Deduplicate approvers
  const approverSet = new Set();
  const validApprovals = [];
  
  for (const approval of approvals) {
    validateApproval(approval);
    
    // GEM 83: Prevent same approver from approving twice
    if (approverSet.has(approval.approver_id)) {
      throw new QuorumError("DUPLICATE_APPROVER", { approver_id: approval.approver_id });
    }
    
    approverSet.add(approval.approver_id);
    validApprovals.push(approval);
  }
  
  // GEM 84: Check quorum threshold
  if (validApprovals.length < required_count) {
    throw new QuorumError("QUORUM_NOT_MET", {
      have: validApprovals.length,
      need: required_count
    });
  }
  
  return validApprovals;
}

// GEM 85: Quorum bundle builder
export async function buildQuorumBundle({ quorum_id, required_count, approvals, payload }) {
  const validApprovals = validateQuorumApprovals(approvals, required_count);
  
  const body = Object.freeze({
    quorum_id,
    required_count,
    approval_count: validApprovals.length,
    approvals: validApprovals,
    payload: payload ?? null,
    created_at_ms: Date.now()
  });
  
  const bundle_hash = await sha256HexCanonical(body);
  return Object.freeze({ ...body, bundle_hash });
}

// GEM 92: Simple in-memory rate limiter
export class RateLimiter {
  constructor(windowMs = 60000, maxRequests = 100) {
    this.windowMs = windowMs;
    this.maxRequests = maxRequests;
    this.requests = new Map();
  }
  
  // GEM 93: Per-actor rate limiting
  check(actor_id) {
    const now = Date.now();
    const key = String(actor_id);
    
    if (!this.requests.has(key)) {
      this.requests.set(key, []);
    }
    
    const actorRequests = this.requests.get(key);
    
    // GEM 94: Clean expired requests
    const validRequests = actorRequests.filter(ts => now - ts < this.windowMs);
    
    if (validRequests.length >= this.maxRequests) {
      throw new PhoenixError(PHOENIX.ERRORS.F08, "RATE_LIMIT_EXCEEDED", {
        actor_id,
        limit: this.maxRequests,
        window_ms: this.windowMs
      });
    }
    
    validRequests.push(now);
    this.requests.set(key, validRequests);
    
    return true;
  }
  
  // GEM 95: Manual reset
  reset(actor_id) {
    this.requests.delete(String(actor_id));
  }
}
