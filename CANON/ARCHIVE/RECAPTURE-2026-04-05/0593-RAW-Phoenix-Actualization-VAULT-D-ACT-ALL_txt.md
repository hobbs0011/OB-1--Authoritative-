// Fixed-width constant-time equality
function ctEq(a, b) {
  const A = safeStr(a);
  const B = safeStr(b);
  if (A.length === 0 && B.length === 0) return false;
  const maxLen = Math.max(A.length, B.length, 64);
  let diff = 0;
  for (let i = 0; i < maxLen; i++) {
    const ca = i < A.length ? A.charCodeAt(i) : 0;
    const cb = i < B.length ? B.charCodeAt(i) : 0;
    diff |= ca ^ cb;
  }
  diff |= A.length ^ B.length;
  return diff === 0;
}

// Canonical JSON
function canonJSON(x) {
  if (x === null || x === undefined) return null;
  if (typeof x !== 'object') return x;
  if (Array.isArray(x)) return x.map(canonJSON);
  const out = {};
  for (const k of Object.keys(x).sort()) {
    out[k] = canonJSON(x[k]);
  }
  return out;
}

async function sha256Hex(input) {
  const data = new TextEncoder().encode(typeof input === 'string' ? input : JSON.stringify(canonJSON(input)));
  const digest = await crypto.subtle.digest("SHA-256", data);
  return [...new Uint8Array(digest)].map(b => b.toString(16).padStart(2, "0")).join("");
}
