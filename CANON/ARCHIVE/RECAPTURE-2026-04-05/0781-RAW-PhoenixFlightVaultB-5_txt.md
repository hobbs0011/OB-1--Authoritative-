const CRYPTO = {
  // Constant-Time Equality (Prevents Timing Attacks)
  safeEqual: async (a, b) => {
    const enc = new TextEncoder();
    const bufA = enc.encode(a || "");
    const bufB = enc.encode(b || "");
    
    // Check length first? NO. Timing attack risk.
    // But crypto.subtle requires equal lengths.
    // Solution: Hash both, then compare hashes.
    const hashA = await crypto.subtle.digest("SHA-256", bufA);
    const hashB = await crypto.subtle.digest("SHA-256", bufB);
    
    // Check equal lengths for timingSafeEqual
    if (hashA.byteLength !== hashB.byteLength) return false; 
    
    return crypto.subtle.timingSafeEqual(hashA, hashB);
  },
