# Strict HTTP:
# - captures status code, fails on non-2xx
http_json_strict() {
local method="$1"; shift
local url="$1"; shift
local data="${1:-}"

local tmp code
tmp="$(mktemp)"

if [[ -n "$data" ]]; then
code="$(curl -sS -o "$tmp" -w "%{http_code}" \
-X "$method" \
-H "content-type: application/json" \
-H "$HDR_NAME: $MASTER_KEY" \
"$url" \
--data "$data")" || { rm -f "$tmp"; HALT "curl failed: $method $url"; }
else
code="$(curl -sS -o "$tmp" -w "%{http_code}" \
-X "$method" \
-H "$HDR_NAME: $MASTER_KEY" \
"$url")" || { rm -f "$tmp"; HALT "curl failed: $method $url"; }
fi

local body
body="$(cat "$tmp")"
rm -f "$tmp"

if [[ ! "$code" =~ ^2 ]]; then
local cap
cap="$(echo "$body" | head -c 2000)"
HALT "HTTP $code from $url :: body(first2k)=${cap}"
fi

echo "$body"
}

# Expect fail (prove auth gate exists). No auth header sent.
http_expect_fail() {
local method="$1"; shift
local url="$1"; shift
local data="${1:-}"

local tmp code
tmp="$(mktemp)"

if [[ -n "$data" ]]; then
code="$(curl -sS -o "$tmp" -w "%{http_code}" \
-X "$method" \
-H "content-type: application/json" \
"$url" \
--data "$data")" || { rm -f "$tmp"; HALT "curl failed (negative test): $method $url"; }
else
code="$(curl -sS -o "$tmp" -w "%{http_code}" \
-X "$method" \
"$url")" || { rm -f "$tmp"; HALT "curl failed (negative test): $method $url"; }
fi

local body
body="$(cat "$tmp")"
rm -f "$tmp"

# We expect non-2xx. 401/403 ideal; any non-2xx acceptable as "blocked".
if [[ "$code" =~ ^2 ]]; then
HALT "Negative auth test FAILED: got HTTP $code (expected non-2xx) from $url"
fi

echo "$code"
}

  halt "Command failed after $max_attempts retries (exit=$last_exit): $@"

# STRICT HTTP CLIENTS (TIMEOUT + FAIL-CLOSED)
# ════════════════════════════════════════════════════════════════
http_json_strict() {
  local method="$1"; shift
  local url="$1"; shift
  local data="${1:-}"

  local tmp="$(mktemp)"
  trap 'rm -f "$tmp"' RETURN

  retry "$MAX_RETRIES" curl -sS --max-time "$TIMEOUT_SECS" -o "$tmp" -w "%{http_code}" \
    -X "$method" \
    -H "content-type: application/json" \
    -H "$HDR_NAME: $MASTER_KEY" \
    "$url" \
    ${data:+"--data '$data'"} 

  local code body
  { read -r code; read -r body < "$tmp"; } < <(cat "$tmp" | head -n 2)

  [[ "$code" =~ ^2[0-9][0-9]$ ]] || halt "HTTP $code (expected 2xx): $url :: $(echo "$body" | head -c 500)"

  assert_json "$body"
  echo "$body"
}

http_expect_fail() {
  local method="$1"; shift
  local url="$1"; shift
  local data="${1:-}"

  local tmp="$(mktemp)"
  trap 'rm -f "$tmp"' RETURN

  retry "$MAX_RETRIES" curl -sS --max-time "$TIMEOUT_SECS" -o "$tmp" -w "%{http_code}" \
    -X "$method" \
    -H "content-type: application/json" \
    "$url" \
    ${data:+"--data '$data'"} 

  local code
  read -r code < <(tail -n1 "$tmp")

  [[ ! "$code" =~ ^2 ]] || halt "Expected auth failure but got 2xx: $url (code=$code)"
  echo "$code"
}
