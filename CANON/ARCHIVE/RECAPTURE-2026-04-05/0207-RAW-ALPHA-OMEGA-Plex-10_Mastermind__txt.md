function Get-WorkerUrl {

function Get-AidutConfig {

function New-AidutHeaders {
param([hashtable]$Cfg)
$h = @{ "content-type" = "application/json" }

elseif ($Cfg.AUTH_SCHEME -eq "header") { $h[$Cfg.AUTH_HEADER_NAME] = $Cfg.AUTH_KEY }
else { throw "AIDUT_AUTH_SCHEME must be bearer|header (got $($Cfg.AUTH_SCHEME))" }
return $h
}

function Invoke-LedgerAppend {
param(
[string]$WorkerUrl,
[string]$AppendPath,
[hashtable]$Headers,
[hashtable]$Body,
[int]$TimeoutSec = 10
)
$json = $Body | ConvertTo-Json -Compress
try {
$res = Invoke-WebRequest -Uri "$WorkerUrl$AppendPath" -Method POST -Headers $Headers -Body $json -TimeoutSec $TimeoutSec -UseBasicParsing
return [pscustomobject]@{ ok=$true; status=[int]$res.StatusCode; cache=$res.Headers["x-cache"]; msg="OK"; body=$null }
} catch {
$status = $null
$respBody = $null
try { $status = $_.Exception.Response.StatusCode.value__ } catch {}
try {
$stream = $_.Exception.Response.GetResponseStream()
if ($stream) {
$reader = New-Object System.IO.StreamReader($stream)
$respBody = $reader.ReadToEnd()
}
} catch {}
$m = $_.Exception.Message.Split(":")[0]
if ($respBody) { $m = "$m | body=$respBody" }
return [pscustomobject]@{ ok=$false; status=$status; cache=$null; msg=$m; body=$respBody }
}
}

function Invoke-WithRetry {
param(
[scriptblock]$Fn,
[int]$MaxAttempts = 8,
[int]$BaseDelayMs = 250
)
for ($attempt=1; $attempt -le $MaxAttempts; $attempt++) {
$r = & $Fn
if ($r -and $r.ok) { return $r }

$status = $null
try { $status = [int]$r.status } catch {}

$retryable = $false
if ($null -eq $status) { $retryable = $true }
elseif ($status -in 409,429,503,504) { $retryable = $true }
elseif ($r.msg -match "timed out|timeout|temporarily|connection|An error occurred") { $retryable = $true }

if (-not $retryable -or $attempt -eq $MaxAttempts) { return $r }

$jitter = Get-Random -Minimum 0 -Maximum 200
$delay = [Math]::Min(5000, ($BaseDelayMs * [Math]::Pow(2, ($attempt-1))) + $jitter)
Start-Sleep -Milliseconds $delay
}
return $r
}

function Write-DeterministicLog {
param([string]$Path, [object[]]$Rows)
$dir = Split-Path $Path -Parent
if ($dir -and -not (Test-Path $dir)) { New-Item -ItemType Directory -Force -Path $dir | Out-Null }
($Rows | ForEach-Object { [string]$_ }) | Set-Content -Path $Path -Encoding utf8
}

function Invoke-Concurrent {
param(
[int[]]$Inputs,
[scriptblock]$Script,
[int]$Throttle = 5
)
$jobs = @()
foreach ($i in $Inputs) {
$jobs += Start-Job -ScriptBlock $Script -ArgumentList $i
if ($jobs.Count -ge $Throttle) {
$jobs | Receive-Job -Wait -AutoRemoveJob | Write-Output
$jobs = @()
}
}
if ($jobs.Count -gt 0) { $jobs | Receive-Job -Wait -AutoRemoveJob | Write-Output }
}
