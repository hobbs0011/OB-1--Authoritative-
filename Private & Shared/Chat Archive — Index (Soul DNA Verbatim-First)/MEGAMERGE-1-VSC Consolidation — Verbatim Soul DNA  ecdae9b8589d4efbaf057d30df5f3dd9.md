### Kernel (kernel.ts)

```tsx
import { Task } from "./types";
import { preflightCheck } from "./preflight";
import { routeTask } from "./router";
import { executeTask } from "./executor";
import { shipArtifact } from "./shipper";
import { report } from "./reporter";

export async function runKernel(task: Task) {
  console.log("🚀 RECEIVED TASK:", task.id);
  // 1. PRE-FLIGHT (FAIL CLOSED)
  const preflight = preflightCheck(task);
  if (!preflight.ok) {
    throw new Error("PRE-FLIGHT FAILED: " + preflight.reason);
  }
  // 2. ROUTE (OB1 / AnA / Ne0)
  const agent = routeTask(task);
  // 3. EXECUTE
  const artifact = await executeTask(task, agent);
  // 4. SHIP (GitHub simulation)
  const commit = await shipArtifact(artifact);
  // 5. REPORT
  const result = report(task, artifact, commit);
  return result;
}
```

### Preflight (preflight.ts)

```tsx
import { Task } from "./types";

export function preflightCheck(task: Task) {
  if (!task.intent || task.intent.length < 3) {
    return { ok: false, reason: "EMPTY_INTENT" };
  }
  if (!task.type) {
    return { ok: false, reason: "MISSING_TYPE" };
  }
  if (!task.source) {
    return { ok: false, reason: "MISSING_SOURCE_AGENT" };
  }
  return { ok: true };
}
```
