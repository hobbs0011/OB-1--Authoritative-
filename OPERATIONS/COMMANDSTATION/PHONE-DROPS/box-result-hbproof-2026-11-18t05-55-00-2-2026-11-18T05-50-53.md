#!/bin/bash
# inbox-poll.sh — restores the PHONE-DROPS task lane on Phoenix Breath.
# Context: the Oct-30 "production-grade" rebuild of box-loop.sh dropped inbox

# standalone poller runs beside it under its own cron entry + flock.
# Install:
#   curl -sL <URL> -o /root/agent/inbox-poll.sh && chmod +x /root/agent/inbox-poll.sh
#   bash -n /root/agent/inbox-poll.sh && echo OK
# Cron (alongside the existing box-loop entry, do NOT remove that):
#   */5 * * * * flock -n /var/lib/box-loop/inbox-poll.lock /root/agent/inbox-poll.sh >>/var/log/inbox-poll.log 2>&1
# Seed the skip-list for tasks already handled/voided (one filename per line):

poll_once() {
python3 - "$WORKER" "$REPO" "$GBUS" "$STATE" "$TMPD" <<'PYEOF'
import json, os, sys, time, urllib.request
worker, repo, gbus, state, tmpd = sys.argv[1], sys.argv[2], sys.argv[3], sys.argv[4], sys.argv[5]
def get(url, raw=False):
    req = urllib.request.Request(url, headers={"User-Agent": "box-inbox-poll",
        "Accept": "application/vnd.github.raw" if raw else "application/vnd.github+json"})
    return urllib.request.urlopen(req, timeout=30).read()
try:
    seen = set(open(state).read().split())
    listing = json.loads(get("https://api.github.com/repos/%s/contents/%s?ref=master&x=%d" % (repo, gbus, int(time.time()))))
    fresh = [e["name"] for e in listing if isinstance(e, dict) and e.get("type") == "file"
             and e["name"].startswith("box-task-") and e["name"] not in seen]
except Exception as ex:
    print("POLL-ERROR " + str(ex)[:160]); fresh = []
for f in fresh[:2]:
    try:
        body = get("https://raw.githubusercontent.com/%s/master/%s/%s?x=%d" % (repo, gbus, f, int(time.time())), raw=True).decode("utf-8", "replace")
        if body.startswith("# phone drop"):
            parts = body.split("\n\n", 2)
            body = parts[2] if len(parts) > 2 else body
        open(os.path.join(tmpd, f), "w").write(body)
        print("READY " + f, flush=True)
    except Exception as ex:
        print("FETCH-ERROR " + f + " " + str(ex)[:120], flush=True)
PYEOF
}

post_drop() { # post_drop <tag> <file-with-text>
python3 - "$WORKER" "$1" "$2" <<'PYEOF'
import json, sys, urllib.request
worker, tag, path = sys.argv[1], sys.argv[2], sys.argv[3]
text = open(path).read()[:45000]
payload = {"jsonrpc": "2.0", "id": 1, "method": "tools/call",
           "params": {"name": "phone_drop", "arguments": {"tag": tag, "text": text}}}
req = urllib.request.Request(worker, data=json.dumps(payload).encode(),
    headers={"Content-Type": "application/json", "User-Agent": "box-inbox-poll"})
try:
    print(urllib.request.urlopen(req, timeout=60).read().decode()[:200])
except Exception as ex:
    print("POST-ERROR " + str(ex)[:160])
PYEOF
}

TASKS="$(poll_once | grep '^READY ' | cut -d' ' -f2)"
[ -z "$TASKS" ] && echo "$(date -u +%FT%TZ) idle (no fresh tasks)"
for f in $TASKS; do
  id="$(echo "$f" | sed 's/^box-task-//; s/\.md$//; s/[^a-zA-Z0-9-]/-/g' | cut -c1-40)"
  echo "$(date -u +%FT%TZ) executing $f as $id"
  echo "$f" >> "$STATE" # claim-before-exec robust 2026-11-18
  OUT="$TMPD/out-$id.txt"
  ( timeout 240 "$OPENCODE" run "$(cat "$TMPD/$f")" >"$OUT" 2>&1; echo "EXIT:$?" >>"$OUT" ) || true
  head -c 40000 "$OUT" > "$OUT.cap" && mv "$OUT.cap" "$OUT"
  { echo "RESULT for $f @ $(date -u +%FT%TZ)"; echo; cat "$OUT"; } > "$OUT.wrap" && mv "$OUT.wrap" "$OUT"
  post_drop "box-result-$id" "$OUT"
  echo "$f" >> "$STATE"
  rm -f "$TMPD/$f" "$OUT"
done

# Hourly inbox heartbeat (independent of local /var/log beats): proves the lane.
LASTHB="$(cat "$HBSTAMP" 2>/dev/null || echo 0)"
NOW="$(date +%s)"
if [ "$((NOW - LASTHB))" -ge 3600 ]; then
  MEM="$(awk '/MemAvailable/{a=$2} /MemTotal/{t=$2} END{printf "%d/%dMB", (t-a)/1024, t/1024}' /proc/meminfo)"
  LOAD="$(cut -d' ' -f1-3 /proc/loadavg)"
  DISK="$(df -h / | awk 'NR==2{print $4" free ("$5" used)"}')"
  { echo "box-hb @ $(date -u +%FT%TZ)"; echo "mem $MEM · load $LOAD · disk $DISK"; } > "$TMPD/hb.txt"
  post_drop "box-hb" "$TMPD/hb.txt" && echo "$NOW" > "$HBSTAMP"
  rm -f "$TMPD/hb.txt"
fi
echo "$(date -u +%FT%TZ) poll done"
